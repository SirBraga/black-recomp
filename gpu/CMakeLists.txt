cmake_minimum_required(VERSION 3.18)
project(black_gpu_bridge LANGUAGES C CXX)
if(APPLE)
 enable_language(OBJCXX)
 add_compile_definitions(VK_USE_PLATFORM_METAL_EXT)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT GS_SOURCE OR NOT RUNTIME_INCLUDE)
 message(FATAL_ERROR "Set GS_SOURCE and RUNTIME_INCLUDE")
endif()
set(PARALLEL_GS_STANDALONE ON CACHE BOOL "" FORCE)
set(GRANITE_ASTC_ENCODER_COMPRESSION OFF CACHE BOOL "" FORCE)
add_subdirectory("${GS_SOURCE}" parallel-gs EXCLUDE_FROM_ALL)
add_library(black-parallel-gs MODULE parallel_gs_module.cpp)
if(APPLE)
 target_sources(black-parallel-gs PRIVATE metal_presenter.mm)
 target_link_libraries(black-parallel-gs PRIVATE "-framework Cocoa" "-framework Metal" "-framework QuartzCore")
endif()
target_include_directories(black-parallel-gs PRIVATE "${RUNTIME_INCLUDE}")
target_link_libraries(black-parallel-gs PRIVATE parallel-gs)
add_executable(black-parallel-module-test ../tests/parallel_gs_module_test.cpp)
target_include_directories(black-parallel-module-test PRIVATE "${RUNTIME_INCLUDE}")
if(APPLE)
 target_sources(black-parallel-module-test PRIVATE ../tests/metal_test_window.mm)
 target_link_libraries(black-parallel-module-test PRIVATE "-framework Cocoa" "-framework Metal" "-framework QuartzCore")
endif()
add_executable(black-parallel-texture-test ../tests/parallel_gs_texture_test.cpp)
target_include_directories(black-parallel-texture-test PRIVATE "${RUNTIME_INCLUDE}")

# CPU swizzle tables provide an independent layout oracle for format reinterpretation.
target_sources(black-parallel-texture-test PRIVATE "${RUNTIME_INCLUDE}/../src/lib/gs/ps2_gs_memory.cpp")
set_property(TARGET black-parallel-texture-test PROPERTY CXX_STANDARD 20)
